# 1. Initialize git repository
git init

# 2. Add all project files (respecting .gitignore to keep .env secrets safe)
git add .

# 3. Check what is staged to make sure .env is not included
git status

# 4. Create your initial commit
git commit -m "Initial commit"

# 5. Rename default branch to main
git branch -M main

# 6. Add your remote repository
git remote add origin https://github.com/YANAMI03/ClassScheduling.git

# 7. Push your code to GitHub
git push -u origin main

# If you get an error like "Updates were rejected because the remote contains work that you don't have locally"
# It means someone else pushed code to GitHub that you don't have yet. To fix this:
# 1. Pull the latest changes from GitHub:
git pull origin main

# 2. Then try pushing again:
git push origin main
